from django.contrib.admin.views.decorators import staff_member_required

2. dodanie dekoatora staff_member_required:

@staff_member_required
def user_list(request):
    users = User.objects.all()

    return render(
        request,
        "users/user_list.html",
        {
            "users": users
        }
    )
