1. dodanie w api/models.py:

class LogEntry(models.Model):
    message = models.TextField()
    created_at = models.DateTimeField(default=timezone.now)

    def __str__(self):
        return f"{self.created_at} - {self.message[:50]}"

2. wykonanie:

python manage.py makemigrations
python manage.py migrate

3. dodanie w api/tasks.py:

from datetime import timedelta

oraz:

@shared_task
def cleanup_old_logs():
    cutoff_date = timezone.now() - timedelta(days=90)

    deleted_count, _ = LogEntry.objects.filter(
        created_at__lt=cutoff_date
    ).delete()

    print(f"Usunięto starych wpisów LogEntry: {deleted_count}")

    return deleted_count

4. dodanie harmonogramu dziennego w config/settings.py:

"cleanup-old-logs-daily": {
    "task": "api.tasks.cleanup_old_logs",
    "schedule": crontab(hour=2, minute=0),
},
