2. utworzenie zadania:

@shared_task
def update_user_last_login(user_id):
    user = User.objects.get(id=user_id)
    user.last_login = timezone.now()
    user.save(update_fields=["last_login"])

    print(
        f"Zaktualizowano last_login użytkownika ID={user_id}: "
        f"{user.last_login}"
    )

    return user_id
