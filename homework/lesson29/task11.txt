1. utworzenie w api/tasks.py zadania:

@shared_task(bind=True)
def progress_task(self):
    for i in range(1, 101):
        time.sleep(0.1)

        self.update_state(
            state="PROGRESS",
            meta={
                "current": i,
                "total": 100,
            },
        )

    return {
        "current": 100,
        "total": 100,
        "status": "Zakończono",
    }

from celery.result import AsyncResult

oraz:

def start_progress_task(request):
    task = progress_task.delay()

    return JsonResponse({
        "task_id": task.id
    })

i:

def task_status(request, task_id):
    task = AsyncResult(task_id)

    response = {
        "task_id": task_id,
        "state": task.state,
    }

    if task.state == "PROGRESS":
        response["current"] = task.info.get("current", 0)
        response["total"] = task.info.get("total", 100)

    elif task.state == "SUCCESS":
        response["current"] = 100
        response["total"] = 100
        response["result"] = task.result

    return JsonResponse(response)
