utworzenie w pliku api/tasks.py zadania multiply, które przyjmuje dwie liczby, mnoży je i zwraca wynik:
@shared_task
def multiply(a, b):
    result = a * b
    print(f"Wynik mnożenia: {a} * {b} = {result}")
    return result
dodanie w pliku api/views.py widoku obsługującego formularz i przekazującego liczby do Celery:
from django.http import HttpResponse
from django.shortcuts import render

from .tasks import hello_world, multiply


def multiply_view(request):
    if request.method == "POST":
        a = int(request.POST.get("a"))
        b = int(request.POST.get("b"))

        multiply.delay(a, b)

        return HttpResponse(
            "Zadanie mnożenia zostało wysłane do Celery."
        )

    return render(request, "multiply.html")
